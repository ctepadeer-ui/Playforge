    Ігор поки немає

    Тут будуть ігри, створені гравцями та збережені в цьому браузері.
    Опиши гру або введи команди ForgeCode. У студії використовується тільки ForgeCode 1.5.

      ForgeCode 1.5 · змінні, умови, цикли, функції, події та команди керування

      Код запускається кнопкою «Грати»
      Довідка ForgeCode
      // Повідомлення та світ
say "Привіт!"
sky "#87ceeb"

// Змінні та властивості
var coins = 0
set coins = coins + 1
set game.score = coins
set player.x = player.x + 1
set me.y = 5

// Умови
if game.score >= 5 {
  say "Рівень пройдено!"
} else if game.score > 0 {
  say "Ще трохи!"
} else {
  say "Збери ще монети"
}

// Повторення (до 1000 разів за один запуск)
repeat 5 {
  coin 0, 2, 0
}

// Функції
function reward {
  give "key", 1
  return
}
call reward

// Події
on touch {
  say "Торкнувся!"
}
on update {
  set me.y = me.y + 0.01
}
on coin {
  say "Монети зібрано"
}

// Об’єкти
block 0, 0, 0, 5, 1, 5, "#00ff00"
coin 0, 2, 0
lava 5, 0, 0, 4, 4
spawn 0, 3, 0
flag 10, 1, 0
finish 20, 1, 0

// Керування
move player 1, 0, 0
teleport 0, 3, 0
rotate me 0, 90, 0
scale me 2, 1, 2
destroy me
damage 10
heal 20
give "key", 1
take "key", 1
var prize = random(1, 100)
